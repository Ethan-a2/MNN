Session API (底层)

操作 Tensor，直接控制推理过程

// 使用流程
Interpreter *interpreter = Interpreter::createFromFile(modelPath);
Session *session = interpreter->createSession(sessionConfig);

// 输入
auto input = session->getInput("input_name");
input->copyFromHostTensor(hostTensor);

// 执行
session->runSession();

// 输出
auto output = session->getOutput("output_name");
output->copyToHostTensor(result);

interpreter->releaseSession(session);
delete interpreter;

Module API (高层)

Express-based 动态图，推荐用于 LLM / Diffusion

// 使用流程
Module::load(loadCache, size, "", "");  // 加载模型

VARP input = _Input({batch, seq}, NCHW);
VARP output = module->forward(input);

// 自动构图、自动调度
std::vector result = output->readMap();
