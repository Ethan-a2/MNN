  static bool __##suite_name##_##test_name(); \

// Gradient checking: analytic vs numerical
TEST(Gradient, Conv2DGradient) {
  // Build expr graph
  Var X = _Input({1, 3, 224, 224});
  Var W = _Weights({64, 3, 3, 3});
  Var Y = _Conv(X, W);
  Var loss = _ReduceMean(Y);
  // Compute analytic gradient
  auto gradExpr = gradient(loss, {X, W});
  // Compute numerical gradient
  auto numGrad = numericalGradient(loss, {X, W}, 1e-4f);
  // Compare
  ASSERT_LE(maxAbsDiff(gradExpr[0], numGrad[0]), 1e-4);
}

Gradient Check Methods

  • Forward-mode AD: Efficient for few inputs
  • Reverse-mode AD: Standard for deep learning
  • Numerical gradient: Finite difference (f(x+h)-f(x))/h
  • Central difference: More accurate (f(x+h)-f(x-h))/(2h)

}

// Simplified test runner
int main(int argc, char** argv) {
  // Iterate all registered test suites
  for (auto& suite : MNNTestSuite::suites()) {
    printf("Running suite: %s\\n", suite.first.c_str());
    // Test on each backend
    for (auto backend : {CPU, OPENCL, METAL, VULKAN, CUDA}) {
      MNNTestSuite::setCurBackend(backend);
      bool passed = suite.second->run();
      logResult(suite.first, backend, passed);
    }
  }
  return 0; // Exit code = #failed tests
}
